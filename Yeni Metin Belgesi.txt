    /* ---------------------------------------------------------------------
       3D SUBWAY SURFERS ENGINE (THREE.JS)

    var scene, camera, renderer, player, obstacles = [], coins = [];
    var currentLane = 0; // -1: Sol, 0: Orta, 1: Sağ
    var targetX = 0;
    var score = 0;
    var gameRunning = false;
    var isJumping = false;
    var jumpVelocity = 0;
    var animFrameId;

    function startSubwayGame() {
        var container = document.getElementById('subwayCanvasContainer');
        container.innerHTML = '<div class="subway-ui">SKOR: <span id="subwayScore">0</span></div><div class="subway-controls">Sol/Sağ Ok Tuşları: Şerit Değiştir | Yukarı Ok: Zıpla</div>';

        score = 0;
        currentLane = 0;
        targetX = 0;
        obstacles = [];
        coins = [];
        gameRunning = true;

        scene = new THREE.Scene();
        scene.background = new THREE.Color(0x11141c);

        camera = new THREE.PerspectiveCamera(60, container.clientWidth / container.clientHeight, 0.1, 1000);
        camera.position.set(0, 3, 6);
        camera.lookAt(0, 1, -5);

        renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        container.appendChild(renderer.domElement);

        var light = new THREE.DirectionalLight(0xffffff, 1);
        light.position.set(0, 10, 5);
        scene.add(light);
        scene.add(new THREE.AmbientLight(0x404040, 1.5));

        // Zemin (Pist)
        var trackGeo = new THREE.PlaneGeometry(6, 200);
        var trackMat = new THREE.MeshBasicMaterial({ color: 0x222531 });
        var track = new THREE.Mesh(trackGeo, trackMat);
        track.rotation.x = -Math.PI / 2;
        track.position.z = -90;
        scene.add(track);

        // Şerit Çizgileri
        for (var l of [-1, 1]) {
            var lineGeo = new THREE.PlaneGeometry(0.1, 200);
            var lineMat = new THREE.MeshBasicMaterial({ color: 0xf39c12 });
            var line = new THREE.Mesh(lineGeo, lineMat);
            line.rotation.x = -Math.PI / 2;
            line.position.set(l * 2, 0.01, -90);
            scene.add(line);
        }

        // Karakter (Player)
        var pGeo = new THREE.BoxGeometry(0.6, 1.2, 0.6);
        var pMat = new THREE.MeshLambertMaterial({ color: 0xf39c12 });
        player = new THREE.Mesh(pGeo, pMat);
        player.position.set(0, 0.6, 0);
        scene.add(player);

        window.removeEventListener('keydown', handleSubwayInput);
        window.addEventListener('keydown', handleSubwayInput);

        animateSubway();
    }

    function handleSubwayInput(e) {
        if (!gameRunning) return;
        if (e.key === "ArrowLeft" && currentLane > -1) {
            currentLane--;
            targetX = currentLane * 2;
        } else if (e.key === "ArrowRight" && currentLane < 1) {
            currentLane++;
            targetX = currentLane * 2;
        } else if (e.key === "ArrowUp" && !isJumping) {
            isJumping = true;
            jumpVelocity = 0.2;
        }
    }

    function spawnObstacle() {
        var lane = Math.floor(Math.random() * 3) - 1;
        var geo = new THREE.BoxGeometry(1.2, 1, 0.8);
        var mat = new THREE.MeshLambertMaterial({ color: 0xe74c3c });
        var obs = new THREE.Mesh(geo, mat);
        obs.position.set(lane * 2, 0.5, -50);
        scene.add(obs);
        obstacles.push(obs);
    }

    function animateSubway() {
        if (!gameRunning) return;
        animFrameId = requestAnimationFrame(animateSubway);

        // Smooth lane change
        player.position.x += (targetX - player.position.x) * 0.2;

        // Jump gravity
        if (isJumping) {
            player.position.y += jumpVelocity;
            jumpVelocity -= 0.012;
            if (player.position.y <= 0.6) {
                player.position.y = 0.6;
                isJumping = false;
            }
        }

        // Spawn obstacles
        if (Math.random() < 0.03) spawnObstacle();

        // Update obstacles
        for (var i = obstacles.length - 1; i >= 0; i--) {
            var obs = obstacles[i];
            obs.position.z += 0.4;

            // Collision detection
            if (Math.abs(obs.position.z - player.position.z) < 0.6 &&
                Math.abs(obs.position.x - player.position.x) < 0.6 &&
                player.position.y < 1.0) {
                gameRunning = false;
                alert("Oyun Bitti! Toplam Skor: " + Math.floor(score));
                startSubwayGame();
                return;
            }

            if (obs.position.z > 5) {
                scene.remove(obs);
                obstacles.splice(i, 1);
            }
        }

        score += 0.1;
        var scoreElem = document.getElementById('subwayScore');
        if (scoreElem) scoreElem.innerText = Math.floor(score);

        renderer.render(scene, camera);
    }

    function stopSubwayGame() {
        gameRunning = false;
        if (animFrameId) cancelAnimationFrame(animFrameId);
        window.removeEventListener('keydown', handleSubwayInput);
    }
